program BasicInput;

var
    n: integer;
    amount: real;
    flag: boolean;
    letter: char;
    word: string;

begin
    readln;
    read(n, amount);
    readln(flag, letter, word);
    writeln('n ', n + 1);
    writeln('amount ', amount);
    writeln('flag ', flag);
    writeln('letter ', letter);
    writeln('word ', word);
end.
