program ArrayByValueParameter;

type
    Index == 1..3;
    Numbers == array[Index] of integer;

var
    nums: Numbers;

procedure Sum(values: Numbers);
begin
    values[2] := 100;
    writeln('array param ', values[1] + values[2] + values[3]);
end;

begin
    nums[1] := 2;
    nums[2] := 5;
    nums[3] := 8;
    Sum(nums);
    writeln('array original ', nums[1] + nums[2] + nums[3]);
end.
