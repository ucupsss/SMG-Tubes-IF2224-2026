program ArrayCopy;

type
    Index == 1..3;
    NumberArray == array[Index] of integer;

var
    source: NumberArray;
    target: NumberArray;

begin
    source[1] := 2;
    source[2] := 5;
    source[3] := 8;
    target := source;
    source[2] := 99;
    writeln('array copy ', target[1] + target[2] + target[3]);
end.
